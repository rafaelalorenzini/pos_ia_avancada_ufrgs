import numpy as np

def transform(grid):
    grid = np.array(grid)
    rows, cols = np.where(grid == 2)
    
    if len(rows) == 0:
        return [[7]]
    
    mean_row = np.mean(rows)
    mean_col = np.mean(cols)
    
    if np.isclose(mean_row, 1.0) and np.isclose(mean_col, 1.0):
        return [[1]]
    else:
        return [[7]]
