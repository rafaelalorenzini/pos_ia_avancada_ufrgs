import numpy as np

def transform(grid):
    grid = np.array(grid)
    H, W = grid.shape
    
    # Find non-zero cells
    non_zeros = []
    for r in range(H):
        for c in range(W):
            if grid[r][c] != 0:
                non_zeros.append((r, c, grid[r][c]))
                
    if len(non_zeros) != 2:
        return grid.tolist()
        
    p1, p2 = non_zeros[0], non_zeros[1]
    r1, c1, v1 = p1
    r2, c2, v2 = p2
    
    delta_r = r2 - r1
    delta_c = abs(c2 - c1)
    
    output = np.zeros((H, W), dtype=int)
    
    if delta_r >= delta_c and delta_c > 0:
        # Horizontal repetition
        P = 2 * delta_c
        row_template = np.zeros(W, dtype=int)
        
        curr_c1 = c1
        while curr_c1 < W:
            row_template[curr_c1] = v1
            curr_c1 += P
            
        curr_c2 = c2
        while curr_c2 < W:
            row_template[curr_c2] = v2
            curr_c2 += P
            
        for r in range(H):
            output[r, :] = row_template
    else:
        # Vertical repetition
        P = 2 * delta_r if delta_r > 0 else 2
        
        curr_r1 = r1
        while curr_r1 < H:
            output[curr_r1, :] = v1
            curr_r1 += P
            
        curr_r2 = r2
        while curr_r2 < H:
            output[curr_r2, :] = v2
            curr_r2 += P
            
    return output.tolist()
