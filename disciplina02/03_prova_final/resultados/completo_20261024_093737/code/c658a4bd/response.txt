import numpy as np

def transform(grid):
    grid = np.array(grid)
    
    # Find all unique non-zero colors
    colors = np.unique(grid)
    colors = colors[colors != 0]
    
    components = []
    for c in colors:
        rows, cols = np.where(grid == c)
        min_r, max_r = rows.min(), rows.max()
        min_c, max_c = cols.min(), cols.max()
        height = max_r - min_r + 1
        width = max_c - min_c + 1
        size = max(height, width)
        components.append((c, size))
        
    # Sort components by size descending
    components.sort(key=lambda x: x[1], reverse=True)
    
    # The output size is the size of the largest component
    S = components[0][1]
    out = np.zeros((S, S), dtype=int)
    
    k = len(components)
    for i, (c, size) in enumerate(components):
        r_start = i
        r_end = S - 1 - i
        c_start = i
        c_end = S - 1 - i
        
        if i == k - 1:
            # Innermost layer: fill the box
            out[r_start:r_end+1, c_start:c_end+1] = c
        else:
            # Outer/intermediate layers: draw border
            out[r_start, c_start:c_end+1] = c
            out[r_end, c_start:c_end+1] = c
            out[r_start:r_end+1, c_start] = c
            out[r_start:r_end+1, c_end] = c
            
    return out.tolist()
