import numpy as np

def transform(grid):
    arr = np.array(grid)
    H, W = arr.shape
    subgrids = []
    
    r = 0
    while r < H - 1:
        cols = []
        for c in range(W - 1):
            if (arr[r, c] == 8 and arr[r, c+1] == 8 and 
                arr[r+1, c] == 8 and arr[r+1, c+1] == 8):
                cols.append(c)
        
        if cols:
            c_left = cols[0]
            c_right = cols[-1]
            subgrid = arr[r:r+2, c_left+2:c_right]
            subgrids.append(subgrid)
            r += 2
        else:
            r += 1
            
    return np.vstack(subgrids).tolist()
